programa
{
	funcao inicio()
	{
		inteiro numero, i, resultado

		faca
		{
			escreva("Digite um número inteiro positivo: ")
			leia(numero)
			
			se (numero <= 0)
			{
				escreva("Erro: O número precisa ser maior que zero!\n\n")
			}
		} enquanto (numero <= 0)

		escreva("\n TABUADA DE ", numero, " \n")
		
				para (i = 1; i <= 10; i++)
		{
			resultado = numero * i
			escreva(numero, " x ", i, " = ", resultado, "\n")
		}
		
	}
}
